cmake_minimum_required(VERSION 3.16)
project(uav_native_planner)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(planner_core)
target_include_directories(planner_core PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} compat)
target_link_libraries(planner_core Eigen3::Eigen nlohmann_json::nlohmann_json)
if(BACKEND STREQUAL "ego")
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(geometry_msgs REQUIRED)
  set(P "${UPSTREAM}/src/planner")
  target_sources(planner_core PRIVATE ego.cpp ${P}/bspline_opt/src/bspline_optimizer.cpp
    ${P}/bspline_opt/src/uniform_bspline.cpp ${P}/bspline_opt/src/gradient_descent_optimizer.cpp
    ${P}/path_searching/src/dyn_a_star.cpp)
  target_include_directories(planner_core PRIVATE ${P}/bspline_opt/include ${P}/path_searching/include ${P}/traj_utils/include)
  ament_target_dependencies(planner_core rclcpp geometry_msgs)
elseif(BACKEND STREQUAL "fast_planner")
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(NLopt REQUIRED)
  set(P "${UPSTREAM}/fast_planner")
  target_sources(planner_core PRIVATE fast.cpp ${P}/bspline_opt/src/bspline_optimizer.cpp ${P}/path_searching/src/kinodynamic_astar.cpp)
  target_include_directories(planner_core PRIVATE ${P}/bspline_opt/include ${P}/path_searching/include ${NLOPT_INCLUDE_DIRS})
  target_link_libraries(planner_core ${NLOPT_LIBRARIES})
  ament_target_dependencies(planner_core rclcpp)
elseif(BACKEND STREQUAL "gcopter")
  # Match the upstream geometry headers' C++14 comparator contract.
  set_property(TARGET planner_core PROPERTY CXX_STANDARD 14)
  target_sources(planner_core PRIVATE gcopter.cpp)
  target_include_directories(planner_core PRIVATE ${UPSTREAM}/gcopter/include)
else()
  message(FATAL_ERROR "Select a real supported upstream core")
endif()
install(TARGETS planner_core DESTINATION bin)
