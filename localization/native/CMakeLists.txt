cmake_minimum_required(VERSION 3.22)
project(uav_slam_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(glim REQUIRED)
find_package(gtsam_points REQUIRED)
find_package(GTSAM REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization filesystem)
add_executable(uav_map_export map_export.cpp)
target_include_directories(uav_map_export PRIVATE ${GLIM_INCLUDE_DIRS} ${GTSAM_INCLUDE_DIR} ${gtsam_points_INCLUDE_DIRS})
target_link_libraries(uav_map_export PRIVATE glim::glim gtsam_points::gtsam_points gtsam Boost::serialization Boost::filesystem)
install(TARGETS uav_map_export DESTINATION bin)
